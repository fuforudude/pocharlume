# Une main sort du manteau pour agripper.
anim attaque 100

cadre jambes-a + corps + objet-frappe + tete decale 2 1 + yeux decale 2 1

cadre jambes + corps decale -1 0 + objet decale -1 0 + tete decale -2 1 + yeux decale -2 1
cadre jambes + corps + objet + tete decale -1 0 + yeux decale -1 0

# Le manteau s'ouvre et s'affaisse : il était plein.
anim mort 160
