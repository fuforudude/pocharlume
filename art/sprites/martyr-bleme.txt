# Les bras s'ouvrent ; la plaie jette sa lumière.
anim attaque 100
cadre jambes + corps decale -1 0 + tete decale -2 0 + yeux decale -2 0 duree 170
cadre jambes-a + corps + tete decale 2 1 + yeux decale 2 1
cadre jambes-a + corps decale 1 0 + tete decale 3 1 + yeux decale 3 1 duree 140
cadre jambes + corps + tete + yeux

anim blesse 120
cadre jambes + corps decale -1 0 + tete decale -2 1 + yeux decale -2 1
cadre jambes + corps + tete decale -1 0 + yeux decale -1 0

# Il s'affaisse doucement, la lueur s'éparpille.
anim mort 160
cadre jambes + corps decale -1 2 + tete decale -3 3 + yeux decale -3 3
cadre jambes-b decale 0 3 + corps decale 0 5 + tete decale -2 7 + yeux decale -2 7
cadre tas duree 900

sprite martyr-bleme-elite
modele martyr-bleme
sombre
